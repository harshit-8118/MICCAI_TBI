import torch
import numpy as np
import wandb
from nnunetv2.training.nnUNetTrainer.nnUNetTrainer import nnUNetTrainer
from nnunetv2.training.loss.deep_supervision import DeepSupervisionWrapper
from nnunetv2.training.loss.dice import MemoryEfficientSoftDiceLoss
from nnunetv2.training.loss.compound_losses import DC_and_CE_loss
import torch.nn as nn


class nnUNetTrainerTBI(nnUNetTrainer):

    def _init_(self, plans: dict, configuration: str, fold: int,
                 dataset_json: dict, device: torch.device = torch.device('cuda')):
        super()._init_(plans, configuration, fold, dataset_json, device)
        self.num_epochs                    = 500
        self.initial_lr                    = 1e-3
        self.weight_decay                  = 3e-5
        self.oversample_foreground_percent = 0.5
        print('nnUNetTrainerTBI: epochs=500, DC+CE loss, oversample=0.5')

    def on_train_start(self):
        super().on_train_start()
        wandb.init(
            project='aims-tbi-segmentation',
            name=f'nnUNetTrainerTBI_fold{self.fold}',
            resume='allow',
            config={
                'fold':       self.fold,
                'epochs':     self.num_epochs,
                'lr':         self.initial_lr,
                'loss':       'DC+CE',
                'oversample': self.oversample_foreground_percent,
                'dataset':    'Dataset002_AIMSTBI_SS',
            }
        )

    def on_train_end(self):
        super().on_train_end()
        wandb.finish()

    def on_epoch_end(self):
        super().on_epoch_end()
        try:
            log_dict = {
                'epoch':      self.current_epoch,
                'lr':         self.optimizer.param_groups[0]['lr'],
                'ema_dice':   self.ema_fg_dice if hasattr(self, 'ema_fg_dice') else 0,
            }
            wandb.log(log_dict)
        except Exception as e:
            print(f'wandb log error: {e}')

    def _build_loss(self):
        # Use proven DC+CE loss — reliable and stable
        loss = DC_and_CE_loss(
            {'batch_dice': self.configuration_manager.batch_dice,
             'do_bg':      False,
             'smooth':     1e-5,
             'ddp':        self.is_ddp},
            {},
            use_ignore_label=self.label_manager.ignore_label is not None,
            dice_class=None
        )
        if self.enable_deep_supervision:
            deep_supervision_scales = self._get_deep_supervision_scales()
            weights = torch.ones(len(deep_supervision_scales))
            weights[1:] = 0.5
            weights = weights / weights.sum()
            loss = DeepSupervisionWrapper(loss, weights)
        return loss
